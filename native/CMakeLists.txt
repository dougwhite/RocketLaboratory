cmake_minimum_required(VERSION 3.20)
project(rocketlaboratory_instrument LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_executable(instrument_core_tests core_tests.cpp)
target_link_libraries(instrument_core_tests PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(instrument_core_tests PRIVATE /W4 /UNDEBUG)
else()
 target_compile_options(instrument_core_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
enable_testing()
add_test(NAME portable_instrument COMMAND instrument_core_tests)
option(ROCKETLAB_BUILD_PLUGIN "Build native SDK adapter; requires licensed MSVC toolchain and external SDK" OFF)
if(ROCKETLAB_BUILD_PLUGIN)
 if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "BakkesMod SDK uses MSVC x64 ABI. MinGW DLL is not a compatible plugin.")
 endif()
 set(BAKKESMOD_SDK "" CACHE PATH "External SDK directory containing include/ and lib/pluginsdk.lib")
 if(NOT EXISTS "${BAKKESMOD_SDK}/lib/pluginsdk.lib")
  message(FATAL_ERROR "Provide external BAKKESMOD_SDK; dependencies are not vendored")
 endif()
 add_library(rocketlab_instrument SHARED plugin.cpp)
 set_property(TARGET rocketlab_instrument PROPERTY CXX_STANDARD 20)
 target_include_directories(rocketlab_instrument PRIVATE "${BAKKESMOD_SDK}/include" "${BAKKESMOD_SDK}/include/bakkesmod/wrappers" "${BAKKESMOD_SDK}/include/bakkesmod/wrappers/items")
 target_link_libraries(rocketlab_instrument PRIVATE "${BAKKESMOD_SDK}/lib/pluginsdk.lib" Threads::Threads)
 target_compile_options(rocketlab_instrument PRIVATE /W4 /EHsc /Zc:__cplusplus)
endif()
